CREATE TABLE product_stocks (
    id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    -- কোন product-এর stock
    product_id BIGINT UNSIGNED NOT NULL,

    -- কোন warehouse/store-এ stock আছে
    warehouse_id BIGINT UNSIGNED NULL,

    -- বর্তমানে মোট কত stock আছে
    quantity DECIMAL(12,2) DEFAULT 0.00,

    -- Order করা হয়েছে কিন্তু এখনও customer-কে দেওয়া হয়নি
    reserved_quantity DECIMAL(12,2) DEFAULT 0.00,

    -- Customer-এর কাছে বিক্রি করার জন্য available stock
    available_quantity DECIMAL(12,2) DEFAULT 0.00,

    -- Low-stock warning level
    min_stock DECIMAL(12,2) DEFAULT 0.00,

    -- Maximum stock level
    max_stock DECIMAL(12,2) DEFAULT 0.00,

    -- নতুন stock আনার প্রয়োজন হলে এই level ব্যবহার হবে
    reorder_level DECIMAL(12,2) DEFAULT 0.00,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    UNIQUE KEY unique_product_warehouse (product_id, warehouse_id)
);
